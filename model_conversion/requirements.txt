# Python packages for the model export. setup_venv.py installs this file first
# and then requirements-arm-tosa.txt (which needs --no-deps). Prefer the
# official CPU-only PyTorch wheels: model export does not need CUDA, and the
# CUDA dependencies can exceed the disk available on clean CI runners.
--extra-index-url https://download.pytorch.org/whl/cpu
#
# executorch, torch and torchao must be mutually compatible. The exact
# executorch pin is written to generated AI layers and reports as the required
# PyTorch::ExecuTorch pack version. The converter verifies that the active
# Python environment contains the same version. Zephyr solutions select their
# external-module pack separately and must keep that selection compatible.
# The export environment supports Python 3.10 to 3.13 (see setup_venv.py).
executorch==1.6.0
torch==2.15.0
torchao==0.18.0

# Vela, the Ethos-U compiler. Same pin as executorch's `ethos_u` extra.
ethos-u-vela==5.2.0
